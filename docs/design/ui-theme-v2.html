evidence chain · assertion: "Creatinine 168 (24 May)"
MATCH (p:Patient {id: $pid})
→ [:HAS_ENCOUNTER]→ (e:Encounter)
→ [:HAS_OBS]→ (o:LabResult)
WHERE o.code = '2160-0' // creatinine
  AND o.effective >= date('2026-05-01')
RETURN o.value, o.unit, o.effective, o.source_ref
ORDER BY o.effective DESC
